Документация

27 скриптов без зависимостей. Всё работает через data-атрибуты; JS API нужен только для особых случаев.

Подключение

Одной строкой, bundle + css или по отдельности (core.min.js + dist/<name>.min.js). Стили (включая Tabs и иконки) — в psyscripts.min.css.

<!-- 1. one line -->
<script src="dist/psyscripts.auto.min.js" defer></script>

<!-- 2. production: styles + bundle -->
<link rel="stylesheet" href="dist/psyscripts.min.css">
<script src="dist/psyscripts.bundle.min.js" defer></script>

<!-- 3. only what you need (core is required for the configuration API) -->
<link rel="stylesheet" href="dist/psyscripts.min.css">
<script src="dist/core.min.js" defer></script>
<script src="dist/tabs.min.js" defer></script>
